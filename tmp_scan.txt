                              <BookOpen className="w-4 h-4 sm:w-4.5 sm:h-4.5 lg:w-5 lg:h-5 text-purple-400 animate-pulse group-hover/viewall:scale-110 transition-transform" />
                              <span className="text-[11px] sm:text-[12px] lg:text-[13px] xl:text-[14px] font-black tracking-[0.18em]">SHOW ALL</span>
                            </motion.button>
                          </div>
                        </div>

                        {/* Scan History Section */}
                        {scanHistory.length > 0 && (
                          <div className="relative z-10 w-full max-w-xl mx-auto px-4 text-left mt-4 mb-4">
                            <div className="flex justify-center mb-4">
                              <button
                                onClick={() => {
                                  setShowScanHistory(!showScanHistory);
                                  sounds.scan();
                                }}
                                className={cn(
                                  hudButtonClass(showScanHistory, 'cyan'),
                                  "px-6 py-3 text-[10px] sm:text-[11px] font-hud font-black uppercase tracking-[0.2em] flex items-center justify-center gap-2 rounded-xl border relative shadow-[0_0_15px_rgba(34,211,238,0.15)] cursor-pointer active:scale-95 transition-all w-full sm:w-auto"
                                )}
                              >
                                <HUDCorners />
                                <Clock className="w-4 h-4 text-cyan-400 shrink-0" />
                                <span>{showScanHistory ? "COLLAPSE SCAN HISTORY" : `VIEW RECENT SCANS (${scanHistory.length})`}</span>
                              </button>
                            </div>

                            <AnimatePresence>
                              {showScanHistory && (
                                <motion.div
                                  initial={{ opacity: 0, height: 0 }}
                                  animate={{ opacity: 1, height: 'auto' }}
                                  exit={{ opacity: 0, height: 0 }}
                                  transition={{ duration: 0.2 }}
                                  className="p-4 bg-slate-950/80 border border-cyan-500/20 rounded-2xl relative shadow-[0_0_20px_rgba(34,211,238,0.05)] overflow-hidden flex flex-col"
                                >
                                  <HUDCorners />
                                  <div className="flex items-center justify-between border-b border-cyan-500/15 pb-2 mb-3 flex-wrap gap-2 shrink-0">
                                    <div className="flex items-center gap-1.5">
                                      <Clock className="w-3.5 h-3.5 text-cyan-400 animate-pulse" />
                                      <h3 className="text-[11px] font-hud font-black text-cyan-400 uppercase tracking-widest">
                                        RECENT SCANS ({scanHistory.length})
                                      </h3>
                                    </div>
                                    {showClearScanConfirm ? (
                                      <div className="flex items-center gap-1.5 bg-slate-900 border border-rose-500/30 px-2 py-0.5 rounded">
                                        <span className="text-[8px] font-mono font-bold text-rose-400 uppercase">Wipe?</span>
                                        <button
                                          onClick={() => {
                                            setScanHistory([]);
                                            localStorage.removeItem('pokethology_scan_history');
                                            setShowClearScanConfirm(false);
                                            try { sounds.scan(); } catch(_) {}
                                          }}
                                          className="px-1.5 py-0.5 text-[8px] font-hud font-black uppercase rounded bg-rose-950 border border-rose-500/40 text-rose-400 hover:bg-rose-900 transition-all cursor-pointer select-none"
                                        >
                                          YES
                                        </button>
                                        <button
                                          onClick={() => {
                                            setShowClearScanConfirm(false);
                                            try { sounds.scan(); } catch(_) {}
                                          }}
                                          className="px-1.5 py-0.5 text-[8px] font-hud font-black uppercase rounded bg-slate-800 border border-slate-700 text-slate-450 hover:bg-slate-750 transition-all cursor-pointer select-none"
                                        >
                                          NO
                                        </button>
                                      </div>
                                    ) : (
                                      <button
                                        onClick={() => {
                                          setShowClearScanConfirm(true);
                                          try { sounds.scan(); } catch(_) {}
                                        }}
                                        className="flex items-center gap-1.5 px-2 py-0.5 text-[8px] font-mono font-bold uppercase rounded border border-rose-900/40 bg-rose-950/20 text-rose-400 hover:bg-rose-900/30 hover:text-rose-300 transition-all cursor-pointer select-none active:scale-95"
                                        title="Clear all scan history"
                                      >
                                        <Trash2 className="w-3.5 h-3.5 text-rose-500" />
                                        Clear All
                                      </button>
                                    )}
                                  </div>
                                  <div className="max-h-[160px] overflow-y-auto custom-scrollbar grid grid-cols-1 sm:grid-cols-2 md:grid-cols-3 gap-2 pr-2">
                                    {scanHistory.map((hist) => (
                                      <div key={hist.name} className="relative group/scanitem">
                                        <button
                                          onClick={() => {
                                            sounds.scan();
                                            performSearch(hist.name);
                                          }}
                                          className="w-full flex items-center gap-2 p-1.5 rounded-xl border border-slate-800 bg-slate-900/40 hover:bg-slate-800 hover:border-slate-600 transition-all text-left"
                                        >
                                          <div className="w-8 h-8 rounded-lg bg-slate-950 flex items-center justify-center shrink-0 border border-slate-800">
                                            <img
                                              src={`https://raw.githubusercontent.com/PokeAPI/sprites/master/sprites/pokemon/${hist.id}.png`}
                                              alt={hist.name}
                                              referrerPolicy="no-referrer"
                                              className="w-7 h-7 object-contain group-hover/scanitem:scale-110 transition-transform"
                                              onError={(e) => {
                                                e.currentTarget.src = 'https://raw.githubusercontent.com/PokeAPI/sprites/master/sprites/pokemon/0.png';
                                              }}
                                            />
                                          </div>
                                          <div className="min-w-0">
                                            <p className="text-[11px] font-sans font-bold uppercase text-slate-200 truncate leading-tight tracking-wider">{hist.name.replace(/-/g, ' ')}</p>
                                            <p className="text-[9px] font-mono text-slate-500 uppercase truncate">#{String(hist.id).padStart(4, '0')}</p>
                                          </div>
                                        </button>
                                        
                                        <div className="absolute bottom-full left-1/2 -translate-x-1/2 mb-2 w-32 opacity-0 invisible group-hover/scanitem:opacity-100 group-hover/scanitem:visible transition-all duration-200 z-50 pointer-events-none scale-95 group-hover/scanitem:scale-100">
                                          <div className="bg-slate-900/95 backdrop-blur-md border border-slate-700 p-2 rounded-xl shadow-xl flex flex-col items-center gap-2 relative">
                                            <div className="absolute -bottom-1.5 left-1/2 -translate-x-1/2 w-3 h-3 bg-slate-900 border-b border-r border-slate-700 rotate-45"></div>
                                            <div className="flex gap-1 flex-wrap justify-center relative z-10">
                                              {hist.types?.map(t => (
                                                <span key={t} className="text-[8px] font-bold uppercase px-1.5 py-0.5 rounded border border-slate-700 bg-slate-800 text-slate-300">
                                                  {t}
                                                </span>
                                              ))}
                                            </div>
                                            <div className="flex items-center gap-1.5 text-[10px] font-mono relative z-10">
                                              <span className="text-slate-400">BST:</span>
                                              <span className={cn("font-bold", (hist.bst || 0) >= 600 ? "text-amber-400" : (hist.bst || 0) >= 500 ? "text-emerald-400" : "text-slate-200")}>
                                                {hist.bst || '???'}
                                              </span>
                                            </div>
                                          </div>
                                        </div>
                                      </div>
                                    ))}
                                  </div>
                                </motion.div>
                              )}
                            </AnimatePresence>
                          </div>
                        )}

                        <div className="absolute bottom-4 left-4 right-4 flex justify-between items-end opacity-40 pointer-events-none">
                          <div className="flex flex-col gap-1">
                            <div className="w-24 h-1 bg-cyan-900/60"></div>
                            <div className="w-16 h-1 bg-cyan-900/60"></div>
                          </div>
                          <div className="text-[8px] font-bold tracking-wider font-mono text-cyan-600 uppercase tracking-widest text-right leading-relaxed select-none">
